# Workflow System Notification
# Script Type: Workflow Task

workflow = frappe.db.get_value(
    "Workflow",
    {"document_type": doc.doctype, "is_active": 1},
    ["name", "workflow_state_field"],
    as_dict=True
)

if workflow:
    state = doc.get(workflow.workflow_state_field or "workflow_state")
    transitions = frappe.get_all(
        "Workflow Transition",
        filters={"parent": workflow.name, "parenttype": "Workflow", "next_state": state},
        fields=["custom_notify_document_creator"]
    )

    recipients = []
    responsible_user = doc.get("custom_current_responsible_user")
    if responsible_user:
        recipients.append(responsible_user)
    if any(row.get("custom_notify_document_creator") for row in transitions) and doc.owner:
        recipients.append(doc.owner)

    notified_users = []
    for user in recipients:
        if user and user not in notified_users and user != frappe.session.user and frappe.db.get_value("User", user, "enabled"):
            notified_users.append(user)
            language = frappe.db.get_value("User", user, "language") or frappe.db.get_single_value("System Settings", "language") or "en"
            document_type = _(str(doc.doctype), lang=language)
            translated_state = _(str(state or "").strip(), lang=language)
            subject = _("{0} requires your action", lang=language).format(document_type)
            message = _(
                "{0} No. {1} has been moved to the workflow state {2}. You are receiving this notification because this document requires your action. Click this notification to open the document, review the details and any available comments, and take the required action.",
                lang=language
            ).format(
                "<strong>" + str(document_type) + "</strong>",
                "<strong>" + str(doc.name) + "</strong>",
                "<strong>" + str(translated_state) + "</strong>"
            )
            frappe.get_doc({
                "doctype": "Notification Log", "subject": subject, "email_content": message,
                "for_user": user, "type": "Alert", "document_type": doc.doctype,
                "document_name": doc.name, "from_user": frappe.session.user, "read": 0
            }).insert(ignore_permissions=True)
